package hua223.calamity.render;

import org.jetbrains.annotations.ApiStatus;

import java.awt.*;
import java.awt.geom.Arc2D;
import java.awt.image.BufferedImage;

//Possible future debugging pie charts and the like
@ApiStatus.Internal
public class JavaGraphicsDebugRenderer {
    public static void drawResourcesPieSlice(PieSlice[] pieSet) {
        BufferedImage img = new BufferedImage(256, 256, BufferedImage.TYPE_INT_ARGB);
        Graphics2D g = img.createGraphics();
        g.setRenderingHint(RenderingHints.KEY_ANTIALIASING, RenderingHints.VALUE_ANTIALIAS_ON);

        double start = 0;
        for (PieSlice s : pieSet) {
            double extent = s.percent * 360.0;
            g.setColor(s.color);
            g.fill(new Arc2D.Double(x, y, w, h, start, extent, Arc2D.PIE));
            g.setColor(Color.BLACK);
            g.draw(new Arc2D.Double(x, y, w, h, start, extent, Arc2D.PIE));
            start += extent;
        }

        g.setColor(Color.WHITE);
        g.drawString("Debug dispatch", 128, 128);
        g.dispose();
    }
}
